Le skater avance tout seul en ligne droite transform.Translate(Vector3.forward * vitesseSkate * Time.deltaTime); // 2. Roulade arrière !"); // Simule la fin de la roulade après 1 seconde Invoke("FinRoulade", 1f); } } void FinRoulade() { estEnTrainDeRouler = false; MettreAJourStatut("Sur le skate"); } public void AfficherPublicite() { MettreAJourStatut("Chargement de la publicité..."); Invoke("PubliciteTermineeRecompense", 2f); // Simule 2 secondes de pub } void PubliciteTermineeRecompense() { MettreAJourStatut("Pub visionnée ! +100 Pièces BTS"); } void MettreAJourStatut(string message) { if (texteStatut != null) texteStatut.text = message; Debug.Log(message); } // Permet d'éviter de déclencher une roulade si on clique sur le bouton Pub private bool EstSurUnBoutonUI(Vector2 positionTactile) { return UnityEngine.EventSystems.EventSystem.current != null && UnityEngine.EventSystems.EventSystem.current.IsPointerOverGameObject(Input.GetTouch(0).fingerId); } }
