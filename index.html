using UnityEngine; using UnityEngine.UI; public class MadMeBTSGame : MonoBehaviour { [Header("Configuration du Jeu : Mad-Me-BTS")] public float vitesseSkate = 5f; private bool estEnTrainDeRouler = false; [Header("Interface UI")] public Text texteStatut; void Start() { MettreAJourStatut("Mad-Me-BTS : Prêt"); } void Update() { // 1. Le skater avance tout seul en ligne droite transform.Translate(Vector3.forward * vitesseSkate * Time.deltaTime); // 2. GESTION DU TACTILE POUR MOBILE if (Input.touchCount > 0) { Touch touche = Input.GetTouch(0); // Dès que le joueur pose le doigt sur l'écran du téléphone if (touche.phase == TouchPhase.Began) { FaireRouladeArriere(); } } // 3. CONTRÔLE POUR PC (Barre d'espace) if (Input.GetKeyDown(KeyCode.Space)) { FaireRouladeArriere(); } } // Fonction de l'esquive public void FaireRouladeArriere() { if (!estEnTrainDeRouler) { estEnTrainDeRouler = true; MettreAJourStatut("Esquive ! Roulade arrière !"); // Simule la fin de la roulade après 1 seconde Invoke("FinRoulade", 1f); } } void FinRoulade() { estEnTrainDeRouler = false; MettreAJourStatut("Sur le skate"); } void MettreAJourStatut(string message) { if (texteStatut != null) texteStatut.text = message; Debug.Log(message); } }
