const container = document.getElementById("game-container"); const skater = document.getElementById("skater"); const obstacle = document.getElementById("obstacle"); const scoreDisplay = document.getElementById("score"); const gameOverText = document.getElementById("game-over-text"); let score = 0; let isJumping = false; let isGameOver = false; let obstacleSpeed = 5; let animationFrameId; // Fonction unique pour gérer l'action (Saut/Roulade ou Recommencer) function handleAction() { if (isGameOver) { resetGame(); } else { jump(); } } // 1. Gestion du CLAVIER (PC) document.addEventListener("keydown", function(event) { if (event.code === "Space") { event.preventDefault(); // Empêche la page de descendre handleAction(); } }); // 2. Gestion du TACTILE (Mobile) container.addEventListener("touchstart", function(event) { event.preventDefault(); // Empêche le zoom ou le scroll mobile instable handleAction(); }, { passive: false }); // 3. Gestion du CLIC SOURIS (PC) container.addEventListener("mousedown", function(event) { if (!isGameOver) jump(); }); function jump() { if (isJumping) return; isJumping = true; skater.classList.add("jump"); setTimeout(() => { skater.classList.remove("jump"); isJumping = false; }, 500); // Durée calée sur l'animation CSS (0.5s) } function gameLoop() { if (isGameOver) return; let containerWidth = container.offsetWidth; let obstacleLeft = parseInt(window.getComputedStyle(obstacle).getPropertyValue("left")); if (obstacleLeft < -25) { obstacleLeft = containerWidth; score++; scoreDisplay.innerText = score; if (score % 5 === 0) { obstacleSpeed += 1; } } obstacle.style.left = (obstacleLeft - obstacleSpeed) + "px"; let skaterBottom = parseInt(window.getComputedStyle(skater).getPropertyValue("bottom")); if (obstacleLeft > 50 && obstacleLeft < 90 && skaterBottom < 40) { endGame(); return; } animationFrameId = requestAnimationFrame(gameLoop); } function endGame() { isGameOver = true; gameOverText.style.display = "block"; cancelAnimationFrame(animationFrameId); } function resetGame() { isGameOver = false; score = 0; obstacleSpeed = 5; scoreDisplay.innerText = score; gameOverText.style.display = "none"; obstacle.style.left = "100%"; gameLoop(); } // Lancement automatique du jeu gameLoop();
